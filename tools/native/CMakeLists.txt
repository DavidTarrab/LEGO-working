cmake_minimum_required(VERSION 3.18)
project(lego_inference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LEGO_UPSTREAM_COMMIT "0a1ef309cdbb9f2024f58a38fb4c42352c5db569")
set(LEGO_LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../local/llama-upstream" CACHE PATH "Pinned upstream source")
execute_process(COMMAND git -C "${LEGO_LLAMA_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE actual_commit OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE git_result)
if(NOT git_result EQUAL 0 OR NOT actual_commit STREQUAL LEGO_UPSTREAM_COMMIT)
    message(FATAL_ERROR "Expected upstream llama.cpp ${LEGO_UPSTREAM_COMMIT}, found ${actual_commit}")
endif()
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(GGML_CUDA ON CACHE BOOL "" FORCE)
# Keep graph replay from bypassing our deliberately selected observation points.
set(GGML_CUDA_GRAPHS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory("${LEGO_LLAMA_SOURCE}" upstream)
add_library(lego-inference SHARED LegoInference.cpp)
target_compile_definitions(lego-inference PRIVATE LEGO_INFERENCE_EXPORTS LEGO_UPSTREAM_COMMIT="${LEGO_UPSTREAM_COMMIT}")
target_link_libraries(lego-inference PRIVATE llama)
target_include_directories(lego-inference PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(lego-infer LegoRunner.cpp)
target_link_libraries(lego-infer PRIVATE lego-inference)
add_executable(lego-gpu-sample LegoGpuSample.cpp)
target_include_directories(lego-gpu-sample PRIVATE "${CUDAToolkit_ROOT}/include")
